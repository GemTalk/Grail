Ultimately the code pattern would be as follows, so that
we do not do further Gci calls within a useraction call after a Gci error has occurred.


class ShimException : public std::exception {
public:
  // could add fields here to hold details , see code examples in server src/topazm.c
  int errNum;
  ShimException() { errNum = 1 };

};

// example low level function
extern "C" PyObject *PyLong_FromSsize_t(Py_ssize_t v) {
   diag_tick(&g_diag_long);
   OopType arg = GciI64ToOop(v);
   OopType res = GciPerformEnv(server, "PyLong_FromSsize_t:", &arg, 1));
   if (res == OOP_ILLEGAL) {
     ShimException ex;
     throw ex;
   }
   return addr_to_pyobj(res);
}


// user actions would contain try ... catch .
static OopType shimCall(...)
{ 
  try {
    ...   shim code
  } catch (ShimException& ex) {
    GciErrSType err;
    if (! GciErr(&err)) 
      assert(err.number != 0);
    }
    if (err.number == 0) {
      err.setError(ERR_Error, "Missing error details from shim code");
    }
    GciRaiseException(&err);
  }
}
